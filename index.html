<script>

/* BANK: all terms + definitions from both reviewer files */

const BANK = [

  { term: "Systems Thinking", definition: "Study and analysis of how interconnected parts work together toward a common purpose; useful for mapping classroom, family, and curriculum interactions." },

  { term: "Linear Thinking", definition: "A sequential cause-and-effect type of thinking that looks for simple one-to-one connections between cause and effect." },

  { term: "Event-Oriented Thinking", definition: "Thinking that views life as a series of events, each with causes and effects; less effective for complex systems." },

  { term: "Financial Management", definition: "Ensuring efficient budgeting, capital allocation, and financial planning." },

  { term: "Leadership and Team Building", definition: "Harnessing collective talent and building effective teams toward shared goals." },

  { term: "Regulatory Compliance", definition: "Ensuring the venture meets laws, regulations, and industry standards." },

  { term: "Impact Measurement", definition: "Evaluating the economic, environmental, and social outcomes of a venture." },

  { term: "Entrepreneurial Behavior", definition: "Behavior shaped by psychological traits like creativity, risk-taking, resilience, cognitive processes, motivation, and environment." },

  { term: "Entrepreneurial Mindset", definition: "A mindset characterized by creativity, adaptability, proactiveness, and opportunity-seeking despite uncertainty." }

/* Question templates that rephrase using the definition */

const TEMPLATES = [

  def => `Which term best matches this definition?\n\n"${def}"`,

  def => `Which concept is described as: "${def}"`,

  def => `Identify the term that fits this description: ${def}`,

  def => `Which of these terms refers to: "${def}"?`,

  def => `What is the correct term for: ${def}`

];


function shuffle(a){

  for(let i=a.length-1;i>0;i--){

    const j=Math.floor(Math.random()*(i+1)); [a[i],a[j]]=[a[j],a[i]];

  }

  return a;

}


/* Create a full question set: one question per BANK item, each with 1 correct + 3 distractors */

function makeQuestion(item){

  const others = BANK.filter(p => p.term !== item.term);

  shuffle(others);

  const distractors = others.slice(0,3).map(o => o.term);

  const options = shuffle([ item.term, ...distractors ]);

  const template = TEMPLATES[Math.floor(Math.random()*TEMPLATES.length)];

  const questionText = template(item.definition);

  return { term: item.term, definition: item.definition, questionText, options, correct: item.term };

}


/* Build & shuffle quiz */

let questionSet = [];

function buildQuiz(){

  const pool = BANK.slice();

  shuffle(pool);

  questionSet = pool.map(p => makeQuestion(p));

  shuffle(questionSet);

}


/* State */

let state = { idx:0, score:0, answered:false, countdownTimer:null, countdownValue:3, overlayVisible:false };


/* DOM refs */

const termEl = document.getElementById('term');

const qEl = document.getElementById('question');

const choicesEl = document.getElementById('choices');

const qIndexEl = document.getElementById('qIndex');

const scoreEl = document.getElementById('score');

const progressBar = document.getElementById('progressBar');

const resultBox = document.getElementById('result');

const feedbackTitle = document.getElementById('feedbackTitle');

const explanationEl = document.getElementById('explanation');

const learnNote = document.getElementById('learnNote');

const restartBtn = document.getElementById('restartBtn');

const playAgain = document.getElementById('playAgain');

const toggleDefinitionBtn = document.getElementById('toggleDefinition');


const overlay = document.getElementById('overlay');

const overlayFeedback = document.getElementById('overlayFeedback');

const overlayDefinition = document.getElementById('overlayDefinition');

const countdownEl = document.getElementById('countdown');

const overlayReset = document.getElementById('overlayReset');

const overlaySkip = document.getElementById('overlaySkip');


function init(){

  buildQuiz();

  state.idx = 0;

  state.score = 0;

  state.answered = false;

  hideOverlayImmediate();

  render();

  updateUI();

}


function updateUI(){

  qIndexEl.textContent = `Question ${state.idx+1} / ${questionSet.length}`;

  scoreEl.textContent = `Score: ${state.score}`;

  const pct = Math.round((state.idx / questionSet.length)*100);

  progressBar.style.width = pct + '%';

}


/* Render current question */

function render(){

  if(state.idx >= questionSet.length){

    showFinal();

    return;

  }

  const q = questionSet[state.idx];

  termEl.textContent = `Term ${state.idx+1}`;

  qEl.textContent = q.questionText;

  resultBox.classList.remove('show');

  resultBox.style.display = 'none';

  explanationEl.textContent = '';

  learnNote.textContent = '';

  nextDisableAllChoices(false);

  choicesEl.innerHTML = '';

  state.answered = false;


  q.options.forEach((opt, i) => {

    const btn = document.createElement('button');

    btn.className = 'choice';

    btn.type = 'button';

    btn.setAttribute('data-choice', opt);

    btn.innerHTML = `<span class="label">${i+1}</span><div class="text">${opt}</div>`;

    btn.addEventListener('click', () => handleSelect(btn, q));

    choicesEl.appendChild(btn);

  });


  updateUI();

}


/* Enable/disable choices' pointer events; also used to prevent further clicks */

function nextDisableAllChoices(disable){

  const items = choicesEl.querySelectorAll('.choice');

  items.forEach(it => {

    it.style.pointerEvents = disable ? 'none' : 'auto';

  });

}


/* When user selects an answer */

function handleSelect(btn, q){

  if(state.answered) return;

  state.answered = true;

  nextDisableAllChoices(true);


  const chosen = btn.getAttribute('data-choice');

  const items = choicesEl.querySelectorAll('.choice');


  // mark chosen + correct

  if(chosen === q.correct){

    btn.classList.add('correct');

    state.score++;

    feedbackTitle.textContent = 'Correct ✓';

    feedbackTitle.style.color = 'var(--correct)';

  } else {

    btn.classList.add('wrong');

    feedbackTitle.textContent = 'Incorrect ✗';

    feedbackTitle.style.color = 'var(--wrong)';

    // highlight correct

    for(const it of items){

      if(it.getAttribute('data-choice') === q.correct) { it.classList.add('correct'); break; }

    }

  }


  // show the inline result area (for immediate reading)

  explanationEl.textContent = `${q.correct}: ${q.definition}`;

  learnNote.textContent = 'Definition shown. Countdown to next question will start.';

  resultBox.classList.add('show');

  resultBox.style.display = 'block';

  updateUI();


  // show overlay with countdown and Reset button — overlay blocks the UI focus

  showOverlay(q);

}


/* Show overlay and start 3-second countdown; overlayReset restarts quiz; overlaySkip jumps to next */

function showOverlay(q){

  overlayFeedback.textContent = feedbackTitle.textContent;

  overlayDefinition.textContent = `${q.correct}: ${q.definition}`;

  countdownEl.textContent = '3';

  overlay.classList.add('show');

  overlay.setAttribute('aria-hidden','false');

  state.countdownValue = 3;

  state.overlayVisible = true;


  // clear any existing timers

  if(state.countdownTimer) { clearInterval(state.countdownTimer); state.countdownTimer = null; }


  state.countdownTimer = setInterval(() => {

    state.countdownValue -= 1;

    if(state.countdownValue <= 0){

      clearInterval(state.countdownTimer);

      state.countdownTimer = null;

      hideOverlayImmediate();

      proceedNext();

    } else {

      countdownEl.textContent = String(state.countdownValue);

    }

  }, 1000);


  // overlay buttons (Reset and Skip) handlers are attached below

}


/* hide overlay right away (used when Reset triggered) */

function hideOverlayImmediate(){

  if(state.countdownTimer){ clearInterval(state.countdownTimer); state.countdownTimer = null; }

  overlay.classList.remove('show');

  overlay.setAttribute('aria-hidden','true');

  state.overlayVisible = false;

}


/* Proceed to the next question (auto after countdown or when skipping) */

function proceedNext(){

  state.idx++;

  if(state.idx >= questionSet.length){

    showFinal();

  } else {

    render();

  }

}


/* Show final screen */

function showFinal(){

  termEl.textContent = 'Round Complete';

  qEl.textContent = `You scored ${state.score} out of ${questionSet.length}.`;

  choicesEl.innerHTML = '';

  progressBar.style.width = '100%';

  resultBox.classList.add('show');

  explanationEl.textContent = 'Replay to get a new randomized order and fresh question phrasing.';

  learnNote.textContent = 'Use Play Again to reshuffle and rehearse.';

  qIndexEl.textContent = 'Completed';

  // store best in localStorage

  const best = Number(localStorage.getItem('study_best') || 0);

  if(state.score > best){

    localStorage.setItem('study_best', state.score);

    // small non-blocking notice

    setTimeout(()=> alert(`New best score: ${state.score}`), 100);

  }

}


/* event wiring */

restartBtn.addEventListener('click', ()=> {

  // Immediate restart (only enabled outside overlay for safety); act as full reset

  if(state.overlayVisible){

    // if overlay visible, ignore; overlayReset handles it

    return;

  }

  init();

});


playAgain.addEventListener('click', ()=> init());


toggleDefinitionBtn.addEventListener('click', ()=>{

  if(resultBox.classList.contains('show')){

    resultBox.classList.remove('show'); resultBox.style.display='none';

  } else {

    resultBox.classList.add('show'); resultBox.style.display='block';

  }

});


/* Overlay Reset (visible during countdown) — restarts whole quiz */

overlayReset.addEventListener('click', ()=> {

  hideOverlayImmediate();

  init();

});


/* Overlay Skip Wait — stops countdown and immediately proceed to next */

overlaySkip.addEventListener('click', ()=> {

  hideOverlayImmediate();

  proceedNext();

});


/* Keyboard support: 1-4 to answer, Enter to skip overlay when visible */

window.addEventListener('keydown', (e)=>{

  if(state.overlayVisible){

    if(e.key === 'Enter'){ // allow quick skip with Enter

      hideOverlayImmediate();

      proceedNext();

    }

    // allow R to reset quickly during overlay

    if(e.key.toLowerCase() === 'r'){

      hideOverlayImmediate();

      init();

    }

    return;

  }

  if(!state.answered && ['1','2','3','4'].includes(e.key)){

    const idx = Number(e.key)-1;

    const btn = choicesEl.children[idx];

    if(btn) btn.click();

  }

});


/* init on load */

init();


</script>
